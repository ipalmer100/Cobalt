# Version metadata stamped into Cobalt.exe.
#
# An unsigned executable carrying no publisher information at all is the
# worst case for Windows' reputation checks, and it also leaves IT with
# nothing to write an allow-list rule against beyond a file hash that
# changes on every build. This does not make the binary trusted -- only a
# code-signing certificate does -- but it gives Explorer's Details tab, the
# SmartScreen prompt and any AppLocker publisher rule something real to
# show.
#
# Update CompanyName to the exact legal entity your IT team expects.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [
          StringStruct('CompanyName', 'Toppan'),
          StringStruct('FileDescription', 'Cobalt - packaging specification editor'),
          StringStruct('FileVersion', '1.0.0.0'),
          StringStruct('InternalName', 'Cobalt'),
          StringStruct('OriginalFilename', 'Cobalt.exe'),
          StringStruct('ProductName', 'Cobalt'),
          StringStruct('ProductVersion', '1.0.0.0'),
        ],
      )
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])]),
  ],
)
